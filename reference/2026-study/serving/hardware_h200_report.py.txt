def select(d,node,**conditions):
    rows=[g for g in d['groups'] if g['node']==node and all(g.get(k)==v for k,v in conditions.items())]
    assert len(rows)==1,(node,conditions,len(rows))
    return rows[0]

def evidence(d):
    compute=[]
    for label,precision in [('FP32 GEMM','fp32'),('TF32 allowed','tf32_allowed'),('FP16','fp16'),('BF16','bf16'),('FP8 E4M3','fp8_e4m3'),('NVFP4','nvfp4'),('FP16 2:4','fp16_sparse_2of4'),('BF16 2:4','bf16_sparse_2of4')]:
        compute.append({'label':label,'nodes':[select(d,n,precision=precision,shape=[8192]*3) for n in [0,1]]})
    scalar=[select(d,n,kind='scalar_fma',threads=256,chains=8,sm_multiplier=4) for n in [0,1]]
    memory=[]
    for kind,mode in [('copy',None),('triad',None),('native_bandwidth','read'),('native_bandwidth','write')]:
        nodes=[]
        for n in [0,1]:
            rr=[g for g in d['groups'] if g['node']==n and g['kind']==kind and abs(g.get('working_set_bytes',0)-2**33)<16 and (g.get('mode')==mode if mode else g.get('stride')==1)]
            assert len(rr)==1,(kind,mode,n,len(rr));nodes.append(rr[0])
        memory.append({'kind':kind,'mode':mode,'nodes':nodes})
    raw=[json.loads(line) for name in ['rdma_results.jsonl','rdma_second_path.jsonl'] for line in (ROOT/'evidence'/name).read_text().splitlines() if line.strip()]
    rdma=[]
    for dev in sorted({x['device'] for x in raw}):
        rr=[x for x in raw if x['device']==dev and x['tool']=='ib_write_bw' and x['client_returncode']==0 and x['server_returncode']==0]
        values=[]
        for x in rr:
            assert '--report_gbits' in x['args'] and not any('use_cuda' in a for a in x['args'])
            line=next(z for z in x['client_stdout'].splitlines() if z.strip().startswith('8388608'))
            values.append(float(line.split()[3]))
        if values:rdma.append({'device':dev,'n':len(values),'gbits_s_median':statistics.median(values),'gb_s_median':statistics.median(values)/8,'range_gbits_s':[min(values),max(values)]})
    collectives=[x for x in d['collectives'] if x['message_bytes']==2**30]
    assert len(collectives)==3
    return {'compute':compute,'scalar':scalar,'memory':memory,'rdma':rdma,'collectives':collectives,'specification_scope':'H200 SXM per GPU; DGX H200 NIC configuration separately identified','summary_sha256':hashlib.sha256((S/'analysis/summary.json').read_bytes()).hexdigest(),'source_manifest':str(E/'sources.json')}

def fmt(g,key):
    s=g[key];return f"{s['median']:.2f} [{s['min']:.2f}–{s['max']:.2f}]; n={s['n']}"
