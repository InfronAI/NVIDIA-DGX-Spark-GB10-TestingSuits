def _classes(tag, *names):
    tag['class'] = list(dict.fromkeys(list(tag.get('class', [])) + list(names)))


def decorate(body, lang):
    soup = BeautifulSoup(body, 'html.parser')
    source_cells = [x.get_text() for x in soup.select('td,th')]
    T = lambda z, e: z if lang == 'zh' else e
    section = 0; heading = ''; counts = defaultdict(int)
    receipt = {'tables': 0, 'figures': 0, 'numeric_cells': 0}
    for tag in soup.find_all(['h2', 'h3', 'h4', 'table', 'figure']):
        if tag.name == 'h2':
            match = re.match(r'(\d+)', tag.get_text(strip=True))
            section = int(match[1]) if match else section
            heading = re.sub(r'^\d+\s*·\s*', '', tag.get_text(' ', strip=True))
        elif tag.name in ('h3', 'h4'):
            heading = tag.get_text(' ', strip=True)
        elif tag.name == 'table':
            counts[(section, 'table')] += 1; receipt['tables'] += 1
            key = f'{section:02d}-{counts[(section, "table")]:02d}'
            _classes(tag, 'result-table')
            headers = tag.select('thead th')
            if len(headers) >= 7: _classes(tag, 'dense-table')
            for h in headers: h['scope'] = 'col'
            caption = soup.new_tag('caption', id='table-caption-' + key)
            number = soup.new_tag('span', attrs={'class': 'visual-label'})
            number.string = T('表 ', 'Table ') + key
            title = soup.new_tag('span', attrs={'class': 'visual-title'}); title.string = heading
            caption.extend([number, title]); tag.insert(0, caption)
            parent = tag.parent
            if 'table-wrap' in parent.get('class', []):
                parent['role'] = 'region'; parent['tabindex'] = '0'
                parent['aria-labelledby'] = caption['id']; parent['id'] = 'table-' + key
            rows = tag.select('tbody tr')
            numeric = defaultdict(int)
            for row in rows:
                cells = row.find_all('td', recursive=False)
                for col, cell in enumerate(cells):
                    text = cell.get_text(' ', strip=True)
                    # Align whole numerical measurements, not prose with embedded numbers.
                    reduced = re.sub(r'(?:US\$|USD|n\s*=|[PTGMK]?FLOP/s|[TGMK]?B/s|GiB|MiB|KiB|GB|MHz|kWh|Wh|W|ms|μs|µs|ns|s|tokens?/s|词元/s|min|years?|年|分钟)', '', text)
                    if col > 0 and re.fullmatch(r'[\d\s.,%+−–—\-→×/／:;；≈<>≤≥()（）\[\]K]+', reduced):
                        _classes(cell, 'numeric'); numeric[col] += 1; receipt['numeric_cells'] += 1
            for col, n in numeric.items():
                if col < len(headers) and n >= max(1, .7 * len(rows)):
                    _classes(headers[col], 'numeric')
        else:
            counts[(section, 'figure')] += 1; receipt['figures'] += 1
            key = f'{section:02d}-{counts[(section, "figure")]:02d}'
            caption = tag.find('figcaption')
            if not caption: continue
            caption['id'] = 'figure-caption-' + key
            label = soup.new_tag('span', attrs={'class': 'visual-label'})
            label.string = T('图 ', 'Figure ') + key
            # Existing scientific note, links and emphasis survive unchanged.
            original = str(caption.decode_contents())
            original = re.sub(r'^(?:图\s*\d+(?:-\d+)?\s*｜|Figure\s+\d+(?:-\d+)?\s*\|\s*)', '', original)
            parts = re.split(r'\s*[｜|]\s*', original, maxsplit=1)
            caption.clear(); caption.append(label)
            title = soup.new_tag('span', attrs={'class': 'visual-title'})
            title.extend(BeautifulSoup(parts[0], 'html.parser').contents); caption.append(title)
            if len(parts) == 2:
                note = soup.new_tag('span', attrs={'class': 'visual-note'})
                note.extend(BeautifulSoup(parts[1], 'html.parser').contents); caption.append(note)
            tag['aria-labelledby'] = caption['id']
            image = tag.find('img')
            if image: image['alt'] = title.get_text(' ', strip=True)
            if image and not image.find_parent(class_='chart-viewport'):
                viewport=soup.new_tag('div',attrs={'class':'chart-viewport','role':'region','tabindex':'0','aria-label':title.get_text(' ',strip=True)})
                image.insert_before(viewport)
                interactive=tag.select_one('.interactive')
                if interactive:viewport.append(interactive.extract())
                viewport.append(image.extract())
    assert source_cells == [x.get_text() for x in soup.select('td,th')], 'Visual styling changed table evidence'
    return str(soup), receipt
