def advance_chapter_references(text):
    """Shift legacy report references after inserting the deployment chapter.

    Only prose chapter references and local section anchors are rewritten;
    scientific quantities, file paths and external source versions stay intact.
    """
    plan_refs=[]
    def preserve(match):
        plan_refs.append(match[0])
        return '\x00planref'+chr(65+len(plan_refs))+'\x00'
    text=re.sub(r'测试计划第\s*\d{1,2}\s*节|\b[Ss]ection\s*\d{1,2}\s+of the test plan',preserve,text)
    def numbers(match):
        def shift(m):
            value=int(m[0]);new=value+1 if value>=2 else value
            return str(new).zfill(len(m[0]))
        return re.sub(r'\d+',shift,match[0])
    text=re.sub(r'第\s*\d{1,2}(?:\s*[、，,/／–—-]\s*\d{1,2})*\s*节',numbers,text)
    text=re.sub(r'\b[Ss]ections?\s*\d{1,2}(?:\s*(?:and|,|/|–|—|-)\s*\d{1,2})*',numbers,text)
    text=re.sub(r'#section-\d{2}\b',numbers,text)
    for i,original in enumerate(plan_refs,1):
        text=text.replace('\x00planref'+chr(65+i)+'\x00',original)
    return text
