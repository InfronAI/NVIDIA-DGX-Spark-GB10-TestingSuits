"""Explicit partial infrastructure-cost scenarios, separate from observed API bills."""
import math
CAPEX_USD=10000.0  # User: two devices at US$5,000 each, 2026-10-15.
ELECTRICITY_USD_KWH=0.109
ELECTRICITY_URL='https://www.globalpetrolprices.com/electricity_prices/'
ELECTRICITY_PERIOD='2026 Q3'

def scenario(tps, power_w, utilization, years=3, hours=8760, capex=CAPEX_USD,
             electricity=ELECTRICITY_USD_KWH, other_annual=0):
    """Constant average provisioned-hour power, zero residual value, no financing."""
    if tps is None or power_w is None or tps<=0 or power_w<0 or not 0<utilization<=1 or years<=0 or hours<=0:return None
    depreciation=capex/years
    energy=power_w/1000*hours*electricity
    generated=tps*3600*hours*utilization
    total=depreciation+energy+other_annual
    return {'years':years,'utilization':utilization,'hours':hours,'tps':tps,'power_w':power_w,
            'depreciation_usd_year':depreciation,'energy_usd_year':energy,
            'other_usd_year':other_annual,'annual_cost_usd':total,
            'generated_million_year':generated/1e6,'usd_per_million_generated':total*1e6/generated,
            'depreciation_share':depreciation/total if total else None}

def scenarios(d):
    rows=[]
    for g in d['natural']:
        resources=[r for r in d['resources'] if r['cell']==g['key']]
        if len(resources)!=2 or any(r['estimated_system_power_mean_w'] is None or not r['power_coverage'] for r in resources):continue
        watts=sum(r['estimated_system_power_mean_w'] for r in resources)
        for years in [2,3,5]:
            for u in [.25,.5,.75,1]:
                rows.append({'cell':g['key'],'sample_power_coverage_min':min(r['power_coverage'] for r in resources),
                             **scenario(g['tps'],watts,u,years)})
    return rows

def break_even_utilization(annual_cost, tps, hours, api_usd_per_million):
    """Accounting parity for the same fixed workload; values >1 are infeasible."""
    if min(tps,hours,api_usd_per_million)<=0:return None
    return annual_cost/(tps*3600*hours/1e6*api_usd_per_million)

def investment(tps, power_w, utilization, price, horizon, hours=8760,
               capex=CAPEX_USD, electricity=ELECTRICITY_USD_KWH,
               other_annual=0, fee_fraction=0):
    """Undiscounted cash return on initial capital; never expense depreciation twice.

    price is avoided API spending or gross realized sales per million generated
    tokens at a fixed input/output mix. utilization must represent useful/sold work.
    No financing, taxes, residual proceeds, replacement or growth are assumed.
    """
    vals=(tps,power_w,utilization,price,horizon,hours,capex,electricity,other_annual,fee_fraction)
    if any(v is None or not math.isfinite(v) for v in vals):return None
    if tps<=0 or power_w<0 or not 0<=utilization<=1 or price<0 or horizon<=0 or not 0<hours<=8760 or capex<=0 or min(electricity,other_annual)<0 or not 0<=fee_fraction<=1:return None
    q=tps*3600*hours*utilization/1e6
    energy=power_w/1000*hours*electricity
    gross=q*price; fees=gross*fee_fraction; operating=energy+other_annual
    net=gross-fees-operating
    cumulative=horizon*net-capex
    required=(capex/horizon+operating)/(q*(1-fee_fraction)) if q>0 and fee_fraction<1 else None
    return {'horizon_years':horizon,'utilization':utilization,'tps':tps,'power_w':power_w,
            'hours':hours,'capex_usd':capex,'price_usd_per_million':price,'fee_fraction':fee_fraction,
            'generated_million_year':q,'energy_usd_year':energy,'other_usd_year':other_annual,
            'gross_benefit_usd_year':gross,'fees_usd_year':fees,'operating_usd_year':operating,
            'net_operating_cash_usd_year':net,'cumulative_net_cash_usd':cumulative,
            'roi_fraction':cumulative/capex,'payback_years':capex/net if net>0 else None,
            'required_price_usd_per_million':required,'recovered_within_horizon':cumulative>=0}

def investment_scenarios(d):
    rows=[]
    for g in d['natural']:
        cost=next((r for r in d['costs'] if r['cell']==g['key']),None)
        power=next((r for r in d['cost_scenarios'] if r['cell']==g['key']),None)
        if not cost or cost['usd_per_million_generated'] is None or not power:continue
        for use,factors in [('self_use',[1]),('sale',[.5,1,2])]:
            for factor in factors:
                for u in [.25,.5,.75,1]:
                    for years in [1,3,5]:
                        rows.append({'cell':g['key'],'use':use,'price_multiple':factor,
                            'api_reference_usd_per_million':cost['usd_per_million_generated'],
                            **investment(g['tps'],power['power_w'],u,
                              cost['usd_per_million_generated']*factor,years)})
    return rows
