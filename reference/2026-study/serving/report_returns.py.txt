"""Self-contained bilingual cash-return scenarios, distinct from measured performance."""
import re

def section(d,lang):
    from build_report import fmt,table
    T=lambda z,e:z if lang=='zh' else e
    rows=d.get('investment_scenarios',[])
    best=max(d['natural'],key=lambda g:g['tps']) if d['natural'] else None
    if not best or not any(r['cell']==best['key'] for r in rows):
        return '\n\n'+T('投资回报测算等待完整的同负载费用与功率依据。','Return scenarios await complete same-workload billing and power evidence.')
    rows=[r for r in rows if r['cell']==best['key']]
    def case(use='self_use',u=.75,years=5,factor=1):
        return next(r for r in rows if (r['use'],r['utilization'],r['horizon_years'],r['price_multiple'])==(use,u,years,factor))
    def money(x):return 'US$'+fmt(x,0)
    def roi(x):return f'{x*100:+.1f}%'
    def pay(r):return T('无正净现金流','No positive net cash flow') if r['payback_years'] is None else fmt(r['payback_years'],2)+T(' 年',' years')
    b=case();c=int(re.search(r'_c(\d+)',best['key'])[1]);p=b['api_reference_usd_per_million']
    text='\n\n<div id="investment-returns"></div>\n\n### '+T('1／3／5 年回本与投资回报：自用和对外售卖','1 / 3 / 5-year payback and investment returns: self-use and sales')
    text+='\n\n'+T('本节回答投入两台设备之后，现金节省或销售收入何时覆盖初始投资。折旧只用于前面的单位成本分摊；这里在购机时一次扣除采购支出，之后只扣现金运营费用，不再重复扣除折旧。所有金额均为美元，以下是基于已测吞吐的条件测算，不是已经实现的多年收益。','This section asks when cash savings or sales receipts cover the initial two-device investment. Depreciation is used above to allocate unit costs; cash-flow analysis instead deducts the purchase once at acquisition and then only cash operating expenses. Depreciation is not deducted again. All amounts are USD. These are conditional projections from measured throughput, not realized multiyear returns.')
    text+=table([T('用途','Use'),T('年度收益如何定义','Definition of annual benefit'),T('价格与工作量口径','Price and work-volume basis')],[
        [T('自用：减少外购 API 支出','Self-use: avoided external API spending'),T('完成同等工作原本需支付的 API 费用','API expense that the same work would otherwise incur'),T('沿用同一输入／输出组合的账户实付综合单价；只计算实际需要的工作量','Same input/output mix and account-paid blended price; only work actually needed counts')],
        [T('售卖：推理服务销售收入','Sales: inference-service receipts'),T('实际售出量 × 综合售价，再扣销售扣费','Sold volume × blended sale price, less sales deductions'),T('展示参考价的 0.5／1／2 倍；有效售出利用率是产生可收款工作量的时间占比','Illustrative 0.5 / 1 / 2× reference price; sale utilization is time producing collectible work')],
    ])
    text+='\n\n'+T(f'**固定测算依据。**采用自然题库并发 {c} 的观测吞吐 {fmt(best["tps"],2)} 词元/s、双机估算功率 {fmt(b["power_w"],2)} W、全年配置 8,760 小时和采购价 US$10,000。对应账户实付综合 API 价格为 **US${fmt(p,5)}/百万生成词元**，已包含这一输入／输出组合的输入费用。售卖部分仅将其用作假设报价的参照，不把账户支出视为节点收入，也不把它视为通用输出词元单价。',f'**Fixed projection basis.** Natural-bank concurrency {c}, observed throughput {fmt(best["tps"],2)} tokens/s, estimated pair power {fmt(b["power_w"],2)} W, 8,760 provisioned hours/year and US$10,000 purchase cost. The corresponding account-paid blended API price is **US${fmt(p,5)} per million generated tokens**, including input fees for this mix. Sales scenarios use it only as a reference for hypothetical quotes; account spending is not node revenue or a general output-token price.')
    text+=table([T('指标','Metric'),T('定义与计算','Definition and calculation')],[
        [T('年有效产出 Q','Annual useful output Q'),T('Q = R × 3,600 × H × u ÷ 1,000,000，单位：百万生成词元','Q = R × 3,600 × H × u / 1,000,000, in million generated tokens')],
        [T('年现金运营费 O','Annual cash operating expense O'),T('O = 双机功率/1,000 × 年配置小时 × 电价 + 其他年度费用；销售扣费单列','O = pair watts/1,000 × provisioned hours × tariff + other annual expense; sales deductions separate')],
        [T('年净现金收益 S','Annual net cash benefit S'),T('自用：S = Q × API 参考价 − O；售卖：S = Q × 售价 × (1 − 扣费比例) − O','Self-use: S = Q × API reference price − O; sales: S = Q × sale price × (1 − deduction rate) − O')],
        [T('第 T 年累计净收益','Cumulative net cash at year T'),T('T × S − K；K 为初始采购投入，负值表示尚未收回全部投入','T × S − K, where K is initial capital; negative means capital is not fully recovered')],
        [T('累计简单投资回报率','Cumulative simple ROI'),T('(T × S − K) ÷ K；分母为初始采购投入，不是营业收入，也不是年化收益率','(T × S − K) / K; denominator is initial capital, not revenue; this is not annualized')],
        [T('简单回本周期','Simple payback'),T('K ÷ S，仅在 S > 0 时有有限值；回本超过 5 年表示五年内未回本','K / S when S > 0; a value over five years means no payback within five years')],
        [T('T 年回本所需最低综合售价','Minimum blended price for T-year payback'),T('(K/T + O) ÷ [Q × (1 − 扣费比例)]；覆盖本节计入的现金费用','(K/T + O) / [Q × (1 − deduction rate)], covering cash costs included here')],
    ])
    text+='\n\n'+T('固定情景假设吞吐、价格和电价在各年保持不变，忽略资金时间价值；不计残值、换机、融资、税务、人工、冷却、机房及其他运维费用。销售扣费基准设为 0，仅用于展示未扣渠道等费用的上界情景，不代表实际免费。自用和售卖各自假设对应的工作需求能够持续实现；有效利用率不是 GPU 活跃度。五年结果还要求设备与服务可持续使用五年。','Fixed scenarios hold throughput, prices and tariffs constant and ignore the time value of money. They exclude residual proceeds, replacement, financing, taxes, staff, cooling, space and other operations. Baseline sales deductions are set to zero solely to illustrate a ceiling before channel and similar fees; zero is not an observed fee. Each use assumes sustained demand for the stated volume; effective utilization is not GPU activity. Five-year results also require five years of usable device and service life.')
    text+='\n\n#### '+T('自用：75% 有效生产比例下的 1／3／5 年现金账','Self-use: 1 / 3 / 5-year cash account at 75% productive utilization')
    text+='\n\n'+T(f'年有效生成量约 {fmt(b["generated_million_year"],1)} 百万词元；年避免 API 支出 {money(b["gross_benefit_usd_year"])}，估算年电费 {money(b["energy_usd_year"])}，年净节省 {money(b["net_operating_cash_usd_year"])}。','Annual useful generation is approximately '+fmt(b['generated_million_year'],1)+' million tokens, avoiding '+money(b['gross_benefit_usd_year'])+' of API expense, with estimated electricity '+money(b['energy_usd_year'])+' and net annual savings '+money(b['net_operating_cash_usd_year'])+'.')
    text+=table([T('持有时长','Holding period'),T('累计避免 API 支出','Avoided API expense'),T('累计估算电费','Cumulative electricity'),T('初始采购','Initial purchase'),T('累计净收益','Net cumulative cash'),T('累计 ROI','Cumulative ROI')],[[T(f'{y} 年',f'{y} year'+('s' if y!=1 else '')),money(case(years=y)['gross_benefit_usd_year']*y),money(case(years=y)['energy_usd_year']*y),money(b['capex_usd']),money(case(years=y)['cumulative_net_cash_usd']),roi(case(years=y)['roi_fraction'])] for y in [1,3,5]])
    text+='\n\n'+T('利用率敏感性：相同的购机与全年用电假设下，只有实际需要的产出才能形成节省。','Utilization sensitivity: with the same purchase and year-round energy assumptions, only useful work creates avoided expenditure.')
    text+=table([T('有效生产比例','Productive fraction'),T('年净节省','Annual net saving'),T('简单回本','Simple payback'),'1 '+T('年 ROI','year ROI'),'3 '+T('年 ROI','year ROI'),'5 '+T('年 ROI','year ROI')],[[f'{u:.0%}',money(case(u=u)['net_operating_cash_usd_year']),pay(case(u=u)),*[roi(case(u=u,years=y)['roi_fraction']) for y in [1,3,5]]] for u in [.25,.5,.75,1]])
    text+='\n\n<!--INVESTMENT_SELF-->\n\n'+T('自用累计现金流与回本｜各线从购机支出 −US$10,000 起算，跨越零线才表示全部收回初始投入。线末百分比为第 5 年累计 ROI。不同线只改变有效生产比例；使用固定短时实测速率外推，线条不是多年实测轨迹。','Self-use cumulative cash and payback | Curves start at the US$10,000 purchase outlay; crossing zero indicates full capital recovery. Endpoint percentages are year-five cumulative ROI. Only productive utilization varies. The curves extrapolate a fixed short-run measured rate; they are not multiyear observations.')
    text+='\n\n'+T(f'**自用结论。**75% 生产比例下，简单回本约 **{pay(b)}**，五年累计 ROI 为 **{roi(b["roi_fraction"])}**；100% 生产比例下，回本约 **{pay(case(u=1))}**，五年 ROI 为 **{roi(case(u=1)["roi_fraction"])}**。其中超过五年的回本值只是同一假设继续成立时的数学外推，不能据此认定设备能持续运营到该年限。',f'**Self-use finding.** At 75% productive utilization, simple payback is **{pay(b)}** and five-year cumulative ROI **{roi(b["roi_fraction"])}**. At 100%, payback is **{pay(case(u=1))}** and five-year ROI **{roi(case(u=1)["roi_fraction"])}**. Payback beyond five years is mathematical extrapolation conditional on unchanged assumptions, not evidence of that device lifetime.')
    text+='\n\n#### '+T('对外售卖：售价与有效售出比例共同决定回报','External sales: price and sold utilization jointly determine return')
    text+='\n\n'+T('下表固定有效售出比例为 75%，以参考价的 0.5／1／2 倍展示报价敏感性。销售额扣除估算电费后为本节年净现金收益；尚未扣除的费用须在实际经营模型中补齐。同价、同工作量且不收销售扣费时，售卖和自用的数值相同，但前者依赖付费客户需求，后者依赖自身真实需求，两种收益不能相加。','The table fixes sold utilization at 75% and varies the quote to 0.5 / 1 / 2 times the reference. Annual net cash here is gross sales less estimated electricity; omitted expenses must be added for actual operations. At equal price and volume with no sales deductions, sales and self-use have equal arithmetic, but sales require paying demand while self-use requires internal demand. Their benefits must not be added together.')
    text+=table([T('售价 USD/百万生成词元','Sale price USD/1M generated'),T('年销售额','Annual gross sales'),T('年净现金收益','Annual net cash'),T('简单回本','Simple payback'),'1 '+T('年 ROI','year ROI'),'3 '+T('年 ROI','year ROI'),'5 '+T('年 ROI','year ROI')],[[f'{p*f:.3f} ({f:g}×)',money(case('sale',factor=f)['gross_benefit_usd_year']),money(case('sale',factor=f)['net_operating_cash_usd_year']),pay(case('sale',factor=f)),*[roi(case('sale',factor=f,years=y)['roi_fraction']) for y in [1,3,5]]] for f in [.5,1,2]])
    twice=case('sale',factor=2)
    text+='\n\n'+T(f'**售卖情景结论。**在 75% 有效售出利用率下，按参考价售卖的回本周期仍为 {pay(b)}；若综合售价为参考价的两倍（US${p*2:.3f}/百万生成词元），回本约 {pay(twice)}，1／3／5 年累计净收益分别为 '+ ' / '.join(money(case('sale',factor=2,years=y)['cumulative_net_cash_usd']) for y in [1,3,5])+'。这说明本情景的回报同时依赖售价与可持续售出量；提高报价本身不能证明市场能够维持该销量。',f'**Sales scenario finding.** At 75% sold utilization, selling at the reference price still gives {pay(b)} payback. At twice that price (US${p*2:.3f}/million generated tokens), payback is about {pay(twice)}, with cumulative net cash at years 1 / 3 / 5 of '+ ' / '.join(money(case('sale',factor=2,years=y)['cumulative_net_cash_usd']) for y in [1,3,5])+'. Return depends on both price and sustained sales volume; a higher quote alone does not establish demand at that volume.')
    text+='\n\n<!--INVESTMENT_SALES-->\n\n'+T('售卖投资回报的价格与利用率敏感性｜三个面板分别是第 1、3、5 年累计简单 ROI；格内为百分比，负值表示尚未收回初始投入。所有面板同一色标，售价是含固定输入配比的综合价，销售扣费与其他运营费均未计入。','Sales ROI sensitivity to price and sold utilization | Panels show cumulative simple ROI at years 1, 3 and 5; negative percentages mean capital is not fully recovered. All panels share one color scale. Prices include the fixed input mix; sales deductions and other operating expenses are excluded.')
    text+='\n\n'+T('指定期限内回本需要多高的综合售价：','Blended sale price required to recover capital within a chosen horizon:')
    text+=table([T('有效售出比例','Sold utilization'),T('1 年回本最低价','1-year payback price'),T('3 年回本最低价','3-year payback price'),T('5 年回本最低价','5-year payback price')],[[f'{u:.0%}',*[f'US${case("sale",u=u,years=y)["required_price_usd_per_million"]:.3f}' for y in [1,3,5]]] for u in [.25,.5,.75,1]])
    text+='\n\n'+T('价格单位均为每百万实际生成词元的综合美元金额，并假设输入／输出配比不变；改变长输入占比、思考长度或缓存条件时，应替换对应负载的吞吐与综合单价再计算。最低售价只覆盖已计入的费用，不等于商业报价建议。','Prices are blended USD per million actual generated tokens at an unchanged input/output mix. Changes in long-input share, reasoning length or cache conditions require replacing both throughput and blended price with matched-workload values. Minimum prices cover only included costs and are not commercial quote recommendations.')
    text+='\n\n<div id="cost-calculator-slot"></div>'
    text+='\n\n['+T('下载全部投资回报情景与计算参数','Download all return scenarios and calculation inputs')+'](investment-scenarios.csv)'
    return text
