class Scan(HTMLParser):
    def __init__(self):
        super().__init__();self.links=[];self.ids=[];self.figures=[];self.h2=[];self.txt=[];self.skip=0;self.heading=False
    def handle_starttag(self,t,a):
        a=dict(a)
        if t in ['script','style']:self.skip+=1
        if 'href' in a:self.links.append(a['href'])
        if 'id' in a:self.ids.append(a['id'])
        if t=='figure':self.figures.append(a.get('id'))
        if t=='h2':self.heading=True
    def handle_endtag(self,t):
        if t in ['script','style']:self.skip-=1
        if t=='h2':self.heading=False
    def handle_data(self,d):
        if not self.skip:self.txt.append(d)
        if self.heading:self.h2.append(d)
