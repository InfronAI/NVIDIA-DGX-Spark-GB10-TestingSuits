def tr(lang, zh, en):
    return zh if lang == 'zh' else en

def tab(lang, headings, rows):
    from build_report import table
    return table([tr(lang, *h) for h in headings], rows)

def figure(token, lang, zh, en):
    return '\n\n<!--'+token+'-->\n\n'+tr(lang, zh, en)

def sources(lang):
    return ' · '.join('['+tr(lang,z,e)+']('+u+')' for z,e,u in SOURCES)

def rdma_rows():
    d=json.loads((ROOT/'evidence/rdma_summary.json').read_text())
    groups={}
    for r in d['rows']:
        if r['client_returncode'] or r['server_returncode'] or not r['data_row']:continue
        parts=r['data_row'].split()
        value=float(parts[3] if r['tool']=='ib_write_bw' else parts[2])
        groups.setdefault((r['device'],r['tool'],r['message_bytes']),[]).append(value)
    return [(device,tool,size,len(values),statistics.median(values),min(values),max(values)) for (device,tool,size),values in groups.items()]
