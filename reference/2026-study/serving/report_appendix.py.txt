def category(path,root_name,relative):
    parts=relative.parts
    if root_name=='shared':return 'shared'
    if root_name=='source':
        return {'datasets':'source_data','sources':'model_sources','test_plan':'source_plan'}.get(parts[0],'source_archive')
    if len(parts)==1 and path.suffix=='.py':return 'code'
    if parts[:2]==('runs','requests'):return 'requests'
    return {'plans':'plans','protocol':'plans','runs':'runs','monitor':'monitor','reports':'results','evidence':'evidence'}.get(parts[0],'runs')

def inventory(a):
    global CACHE
    if CACHE is not None and CACHE['performance_snapshot_utc']==a['generated_utc']:return CACHE
    started=utc();rows=[]
    def add(p,root_name,relative):
        if p.name.startswith('artifact-index') or p.name in {'.DS_Store'} or p.suffix in {'.tmp','.lock','.pyc'}:return
        if p.name=='.env' or p.name.startswith('.env.') or p.suffix in {'.token','.key','.pem'}:return
        if p.is_symlink():return
        try:stat=p.stat()
        except FileNotFoundError:return
        row={'category':category(p,root_name,relative),'root':root_name,'absolute_path':str(p),
             'size_bytes':stat.st_size,'modified_utc':dt.datetime.fromtimestamp(stat.st_mtime,dt.timezone.utc).isoformat(),
             'sha256':hashlib.sha256(p.read_bytes()).hexdigest() if p.suffix in {'.py','.js','.cjs','.mjs','.sh','.go'} else ''}
        rows.append(row)
    for name,root in [('current',ROOT),('source',OLD)]:
        for parent,dirs,files in os.walk(root):
            dirs[:]=sorted(x for x in dirs if x not in {'__pycache__','.git','.venv','node_modules'})
            for filename in sorted(files):
                p=Path(parent)/filename;add(p,name,p.relative_to(root))
    for p in SHARED:
        if p.is_file():add(p,'shared',p.relative_to(REPO))
    rows.sort(key=lambda r:(r['root'],r['category'],r['absolute_path']))
    result={'scan_started_utc':started,'scan_finished_utc':utc(),'performance_snapshot_utc':a['generated_utc'],
      'roots':{'current':str(ROOT),'source':str(OLD),'shared':str(REPO)},'file_count':len(rows),
      'scope':'Current experiment tree and source tree recursively; explicitly listed shared source/dependency files. Index artifacts, bytecode, dependency installations, lock/temp files and credentials excluded.',
      'consistency':'File metadata observed during scan; live files may grow. Inventory inclusion does not imply experimental completion. Source-code hashes are computed; raw data sizes are metadata, not content hashes.',
      'files':rows}
    dump(OUT/'artifact-index.json',result)
    path=OUT/'artifact-index.csv';temp=path.with_suffix('.csv.tmp')
    with temp.open('w',encoding='utf-8-sig',newline='') as f:
        writer=csv.DictWriter(f,fieldnames=['category','root','absolute_path','size_bytes','modified_utc','sha256']);writer.writeheader();writer.writerows(rows)
    temp.replace(path)
    for lang in ['zh','en']:write_browser(result,lang)
    CACHE=result
    return result
