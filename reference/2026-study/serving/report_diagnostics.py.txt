def number(value):
    try:
        v = float(value)
        return v if math.isfinite(v) else None
    except (TypeError, ValueError): return None

def cost_for_requests(requests, billing_rows, quota_per_usd):
    """Require exact identifiers, one terminal success and complete count coverage."""
    by_id = collections.defaultdict(list)
    for row in {str(x['id']): x for x in billing_rows}.values():
        by_id[row['request_id']].append(row)
    matched = []
    for r in requests:
        rows = by_id.get(r.get('request_id'), [])
        if not rows or sum(x['type'] == 2 for x in rows) != 1: continue
        if any(x['model_provider_id'] != 2084 or x['channel_id'] != 2538 for x in rows): continue
        matched.append((r, sum(x['quota'] for x in rows) / quota_per_usd))
    known = len(matched) == len(requests) and bool(requests)
    outputs = [number(r.get('completion_tokens')) for r, _ in matched]
    inputs = [number(r.get('prompt_tokens')) for r, _ in matched]
    output_total = sum(outputs) if known and all(x is not None for x in outputs) else None
    input_total = sum(inputs) if known and all(x is not None for x in inputs) else None
    cost = sum(c for _, c in matched) if known else None
    return {'requests': len(requests), 'matched': len(matched), 'cost_usd': cost,
            'input_tokens': input_total, 'output_tokens': output_total,
            'usd_per_1000_responses': cost * 1000 / len(requests) if known else None,
            'usd_per_million_generated': cost * 1e6 / output_total if cost is not None and output_total else None,
            'request_ids': [r['id'] for r in requests]}

def window_measure(rows, windows, key, max_gap=20):
    """Left-held integration; never bridge a missing value or >20-second gap."""
    rows = sorted(rows, key=lambda r: r['epoch']); times = [r['epoch'] for r in rows]
    duration = sum(end-start for start, end in windows)
    area = covered = 0.; values = []
    for start, end in windows:
        values.extend(number(r.get(key)) for r in rows if start <= r['epoch'] <= end and number(r.get(key)) is not None)
        i = max(0, bisect.bisect_right(times, start)-1)
        while i+1 < len(rows) and times[i] < end:
            left, right = rows[i], rows[i+1]
            value = number(left.get(key)); following = number(right.get(key))
            overlap = max(0., min(end, right['epoch'])-max(start, left['epoch']))
            gap = right['epoch']-left['epoch']
            if 0 < gap <= max_gap and overlap and value is not None and following is not None:
                area += value*overlap; covered += overlap
            i += 1
    return {'mean': area/covered if covered else None, 'max': max(values) if values else None,
            'covered_s': covered, 'window_s': duration,
            'coverage': covered/duration if duration else None}

def normalized_cost(tps, utilization, hourly_cost=1):
    if not tps or not 0 < utilization <= 1: return None
    return hourly_cost * 1e6 / (3600 * tps * utilization)

def csv_export(path, rows):
    if not rows:
        path.write_text(''); return
    keys = list(dict.fromkeys(k for r in rows for k in r if not isinstance(r[k], (list, dict))))
    with path.open('w') as f:
        writer = csv.DictWriter(f, fieldnames=keys); writer.writeheader()
        writer.writerows({k: r.get(k) for k in keys} for r in rows)

def build(a):
    with (ROOT/'reports/requests.csv').open() as f: rows = list(csv.DictReader(f))
    v = presentation(a)
    billing_path = ROOT/'evidence/billing_latest.json'
    billing_bytes = billing_path.read_bytes(); billing = json.loads(billing_bytes)
    costs = []; resources = []; lengths = []
    for g in v['groups']:
        ids = set(g['block_ids']); rs = [r for r in rows if r['block'] in ids and r['state']=='received' and not r.get('retry_of')]
        cost = cost_for_requests(rs, billing['rows'], billing['quota_per_usd'])
        assert len(rs) == g['n'], 'Cohort mismatch'
        costs.append({'cell':g['key'], 'n':g['n'], 'blocks':g['blocks'], 'tps':g['tps'],
                      'ttft_p95_s':g['ttft']['p95'], **cost})
        m = re.fullmatch(r'P2_general_i(\d+)_o(\d+)_c(\d+)',g['key'])
        if m:
            lengths.append({'cell':g['key'], 'input_target':int(m[1]), 'output_cap':int(m[2]),
                'concurrency':int(m[3]), 'n':len(rs), 'blocks':g['blocks'], 'tps':g['tps'],
                'ttft_median_s':g['ttft']['p50'], 'e2e_median_s':g['e2e']['p50'],
                'tpot_median_ms':g['tpot_ms']['p50'],
                'actual_input_mean':describe([number(r['prompt_tokens']) for r in rs if number(r['prompt_tokens']) is not None])['mean'],
                'actual_output_mean':g['output']['mean'], 'truncated':sum(r['truncated']=='True' for r in rs)})
        if not g['key'].startswith('P1_'): continue
        windows = [(b['start_epoch'],b['end_epoch']) for b in v['eligible_blocks'] if b['id'] in ids]
        for node in [0,1]:
            samples = [r for r in a['monitoring']['hardware'] if r['node']==node]
            measured = {key:window_measure(samples,windows,key) for key in
                ['gpu_percent','temperature_c','gpu_power_w','estimated_system_power_w','thermal_throttle']}
            resources.append({'cell':g['key'],'node':node,'concurrency':int(re.search(r'_c(\d+)',g['key'])[1]),
                'gpu_mean_percent':measured['gpu_percent']['mean'], 'gpu_coverage':measured['gpu_percent']['coverage'],
                'temperature_peak_c':measured['temperature_c']['max'],
                'sensor_power_mean_w':measured['gpu_power_w']['mean'],
                'estimated_system_power_mean_w':measured['estimated_system_power_w']['mean'],
                'power_coverage':measured['estimated_system_power_w']['coverage'],
                'throttle_time_fraction':measured['thermal_throttle']['mean'],
                'window_s':sum(end-start for start,end in windows)})
    behavior=[]
    for cell in ['P1_c1_none','P3_c1_low','P1_c4_none','P3_c4_low']:
        rs=[r for r in rows if r['cell']==cell and r['state']=='received' and r['warmup']!='True']
        if not rs: continue
        outputs=[number(r['completion_tokens']) for r in rs if number(r['completion_tokens']) is not None]
        behavior.append({'cell':cell, 'n':len(rs),'output_n':len(outputs),'output_mean':describe(outputs)['mean'],
            'truncated':sum(r['truncated']=='True' for r in rs),
            'no_visible':sum(number(r.get('first_visible_s')) is None for r in rs)})
    nonwarm=[b for b in a['blocks'] if not b.get('warmup') and b.get('cell')]
    result={'generated_utc':a['generated_utc'], 'billing_utc':billing['utc'],
        'billing_source_sha256':hashlib.sha256(billing_bytes).hexdigest(), 'costs':costs,'lengths':lengths,'resources':resources,
        'behavior':behavior, 'natural':[g for g in v['groups'] if g['key'].startswith('P1_')],
        'sample_flow':{'attempts':a['total_sent'],'terminal':len(rows),'complete':sum(r['state']=='received' for r in rows),
            'measured_complete':sum(r['state']=='received' and r['warmup']!='True' for r in rows),
            'eligible_responses':sum(g['n'] for g in v['groups']),'recorded_blocks':len(nonwarm),
            'eligible_blocks':len(v['eligible_blocks']),'excluded_blocks':len(v['excluded_blocks'])},
        'cost_definition':'All charges for exact eligible client-success request IDs / original generated output; input charges included. Missing bills are not zero.',
        'resource_definition':'Time-weighted within the same natural complete-block windows; <=20-second sample gaps only; coverage explicit.',
        'tco_inputs':{'purchase_cost':CAPEX_USD,'currency':'USD','electricity_per_kwh':ELECTRICITY_USD_KWH,
          'electricity_source':ELECTRICITY_URL,'electricity_period':ELECTRICITY_PERIOD,'amortization_years':[2,3,5],
          'provisioned_hours_year':8760,'residual_value':0,'other_costs_included':False},
        'scenario_definition':'Device depreciation plus estimated electricity only; constant sampled systemDraw sum held across provisioned hours; productive fraction 0.25/0.50/0.75/1.00 and fixed observed throughput. Neither measured outlet energy nor full TCO.'}
    result['cost_scenarios']=scenarios(result)
    result['investment_scenarios']=investment_scenarios(result)
    result['investment_definition']='Undiscounted cash return on initial US$10,000; horizons 1/3/5 years. Self-use benefit is avoided same-cohort API spend. Sales use illustrative 0.5/1/2 times that blended price, not observed revenue. Constant useful/sold output and constant estimated provisioned-hour power; no depreciation deducted from cash again. Baseline excludes fees, other operating costs, tax, financing, replacement and residual proceeds.'
    dump(ROOT/'reports/diagnostics.json',result)
    for name,rs in [('cohort-costs',costs),('cohort-resources',resources),('length-metrics',lengths),('cost-scenarios',result['cost_scenarios']),('investment-scenarios',result['investment_scenarios'])]:
        csv_export(ROOT/'reports'/f'{name}.csv',rs)
    return result
