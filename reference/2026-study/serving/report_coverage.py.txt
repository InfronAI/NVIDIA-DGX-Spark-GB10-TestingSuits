"""Final evidence coverage, kept separate from executor termination and call caps."""
from pathlib import Path
import collections
import hashlib
import json

from common import ROOT

OUT=ROOT/'evidence/coverage_clarification_20261017'

def audit():
    paths={'initial':ROOT/'reports/analysis.json','length':ROOT/'reports/length_analysis.json',
           'hardware':ROOT/'hardware_unloaded_confirmed_20261016/analysis/summary.json',
           'completion':ROOT/'gap_completion_20261017/analysis/summary.json',
           'public_plan':ROOT/'gap_completion_20261017/public/manifest.json'}
    a,l,h,d,p=(json.loads(paths[x].read_text()) for x in ['initial','length','hardware','completion','public_plan'])
    measured=sum(v[suite].get('measured',0) for v in h['coverage_by_suite'].values() for suite in ['base','extensions'])
    cache=collections.Counter(x['pair'] for x in d['cache'])
    counters={(x['node'],x['mode'],x['round']) for x in d['counters']}
    continuous=[x for x in d['public']['blocks'] if x.get('complete_continuous_window')]
    rows=[
        ('单节点硬件轮次','Single-node hardware rounds',measured,3740),
        ('双机集合通信','Two-node collective trials',h['paired_collective_trials'],840),
        ('计算与通信重叠','Compute/communication pairs',h['paired_overlap_trials'],120),
        ('原生固定长输出','Native fixed-output requests',sum(x['state']=='complete' for x in d['native_requests']),len(d['native_requests'])),
        ('GPU剖析配对','GPU profiling pairs',len(d['profile_pairs']),9),
        ('64K缓存配对','64K cache pairs',sum(n==2 for n in cache.values()),3),
        ('FP64矩阵轮次','FP64 matrix rounds',len(d['fp64']),30),
        ('代表性计数器','Counter profiles',len(counters),12),
        ('公网连续一小时','Continuous public hour',len(continuous),1)]
    pairs=d['public']['interference'];planned=p['interference_pairs']
    completed=sum(x['fully_observed'] for x in pairs)
    result={'execution_complete':d['completion_ready'],'full_evidence_coverage':False,
            'definition':'Terminal execution is distinct from valid measurement coverage; conditional request caps are not mandatory sample counts.',
            'fixed_coverage':[{'zh':z,'en':e,'valid':v,'planned':n} for z,e,v,n in rows],
            'public_length':{'planned':l['planned'],'sent':l['sent'],'complete':l['complete'],
                             'noncomplete_sent':l['sent']-l['complete'],'not_sent':l['planned']-l['sent'],
                             'exact_length_qualified':l['qualified']},
            'interference':{'candidate_pairs':planned,'executed_pairs':len(pairs),'fully_observed_pairs':completed,
                            'incomplete_pairs':len(pairs)-completed,'not_admitted_pairs':planned-len(pairs),
                            'incomplete_reason':'Short-request local admission rejection; neither arm may be imputed.',
                            'not_admitted_reason':'32 fixed-rate arrivals did not fit the 180-second candidate window.'},
            'continuous_hour':[{'offered':x['offered'],'sent':x['sent'],'complete':x['complete'],
                                'local_rejections':x['local_rejections'],'continuous_s':x['continuous_s']} for x in continuous],
            'resource_limits':d['resource_limits']+['Strict per-token timestamps unavailable from aggregated SSE events'],
            'initial_phases':a['phase_counts'],
            'source_sha256':{str(path.relative_to(ROOT)):hashlib.sha256(path.read_bytes()).hexdigest() for path in paths.values()}}
    return result

def historical_table(a,lang):
    from build_report import table
    from report_labels import phase_name
    T=lambda z,e:z if lang=='zh' else e
    rows=[]
    for p in a['phase_counts']:
        complete=p['complete']+p['expected_context_rejections'];sent=p['logical_sent']
        basis=T('条件上限','Conditional cap') if p['phase'] in ['P7','P8'] else T('固定计划','Fixed plan')
        rows.append([phase_name(p['phase'],lang),f'{p["cap"]:,} ({basis})',sent,complete,sent-complete])
    return table([T('初始公网批次','Initial public batch'),T('设计数量的含义','Design count and meaning'),T('已发送逻辑请求','Logical requests sent'),T('完整／预期边界结果','Complete / expected boundary outcome'),T('未完整结果','Incomplete outcomes')],rows)

def draw(lang,save):
    import matplotlib.pyplot as plt
    from matplotlib.patches import Patch
    T=lambda z,e:z if lang=='zh' else e
    d=audit();OUT.mkdir(parents=True,exist_ok=True)
    (OUT/'audit.json').write_text(json.dumps(d,ensure_ascii=False,indent=2))
    fig,(ax,right)=plt.subplots(1,2,figsize=(12.7,5.7),gridspec_kw={'width_ratios':[1.15,1]})
    blue='#004AB9';amber='#A87400';grey='#D8E2F0'
    for n,r in enumerate(d['fixed_coverage']):
        pct=100*r['valid']/r['planned']
        ax.barh(n,100,color=grey,height=.55);ax.barh(n,pct,color=blue,height=.55)
        ax.text(103,n,f'{r["valid"]:,}/{r["planned"]:,}',va='center',fontsize=8)
    ax.set_yticks(range(len(d['fixed_coverage'])),[r[lang] for r in d['fixed_coverage']]);ax.invert_yaxis()
    ax.set_xlim(0,155);ax.set_xticks([0,50,100]);ax.set_xlabel(T('有效轮次／固定计划（%）','Valid units / fixed plan (%)'))
    ax.set_title(T('已完成的固定实验清单','Completed fixed experiment lists'))
    l=d['public_length'];i=d['interference']
    parts=[(l['complete'],l['noncomplete_sent'],l['not_sent']),
           (i['fully_observed_pairs'],i['incomplete_pairs'],i['not_admitted_pairs'])]
    for n,values in enumerate(parts):
        left=0
        for value,color in zip(values,[blue,amber,grey]):
            width=100*value/sum(values);right.barh(n,width,left=left,color=color,height=.35);left+=width
    right.set_yticks([0,1],[T('公网长度：响应完整性','Public length: response completeness'),T('完整干扰配对','Complete interference pairs')])
    right.set_ylim(1.7,-.65);right.set_xlim(0,100);right.set_xticks([0,50,100]);right.set_xlabel(T('同一清单内的证据状态（%）','Evidence status within each list (%)'))
    right.set_title(T('仍有限制的实验覆盖','Remaining coverage limitations'))
    right.text(0,.37,T(f'{l["complete"]}完整 / {l["noncomplete_sent"]}未完整 / {l["not_sent"]}未发送',f'{l["complete"]} complete / {l["noncomplete_sent"]} incomplete / {l["not_sent"]} unsent'),fontsize=8)
    right.text(0,1.37,T(f'{i["fully_observed_pairs"]}完整 / {i["incomplete_pairs"]}不完整 / {i["not_admitted_pairs"]}未准入',f'{i["fully_observed_pairs"]} complete / {i["incomplete_pairs"]} incomplete / {i["not_admitted_pairs"]} not admitted'),fontsize=8)
    fig.legend(handles=[Patch(color=blue,label=T('完成（按各行定义）','Complete (as defined per row)')),Patch(color=amber,label=T('已执行但未完整','Executed, incomplete')),Patch(color=grey,label=T('未执行','Not executed'))],loc='lower center',ncol=3,bbox_to_anchor=(.5,-.03),frameon=False)
    fig.suptitle(T('最终证据覆盖：各行分母不同，不合成总完成率','Final evidence coverage: distinct row denominators, no overall completion percentage'),fontsize=11)
    save(fig,'workflow',lang)
