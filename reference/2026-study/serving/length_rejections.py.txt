"""Exact terminal length-probe rejections are observations, never replay permission."""
import json
from common import dump,utc,digest

def review(root,results,rows):
 manifest=root/'plans/length_expansion/request_index.json'
 if not manifest.exists():return set()
 index=json.loads(manifest.read_text());verified={}
 for r in results:
  if r['id'] not in index or r.get('state')!='http_error' or r.get('status') not in [400,422]:continue
  rid=r.get('response_request_id') or r.get('request_id');b=[x for x in rows if x['request_id']==rid]
  if len([x for x in b if x['type']==6])!=1 or not all(x['type'] in [6,7] and x['model_provider_id']==2084 and x['channel_id']==2538 and json.loads(x['other']).get('status_code')==r['status'] for x in b):continue
  c=index[r['id']];path=root/'runs/requests'/r['id'];request=json.loads((path/'request.json').read_text())
  natural=(root/'protocol/natural_length_amendment.json').exists() and c['mode']=='fixed' and not c['id'].startswith('length_calibration') and 'ignore_eos' not in request
  if request.get('max_tokens')!=c['max_tokens'] or (not natural and request.get('ignore_eos')!=c['ignore_eos']):continue
  verified[rid]=dict(case_id=r['id'],status=r['status'],request_sha256=digest(path/'request.json'),result_sha256=digest(path/'result.json'),log_ids=[x['id'] for x in b],outcome='terminal_rejection; exact reason not inferred; no replay; later output tiers remain gated')
 dump(root/'evidence/length_rejections.json',dict(utc=utc(),cases=verified));return set(verified)
