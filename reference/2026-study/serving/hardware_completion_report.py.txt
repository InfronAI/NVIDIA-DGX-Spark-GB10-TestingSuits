def tr(lang,z,e):
    text=z if lang=='zh' else e
    return text.replace('../hardware_completion_20261016/', '../'+S.name+'/') if UNLOADED else text

def data():
    p=S/'analysis/summary.json'
    return json.loads(p.read_text()) if p.exists() else None

def table(lang,headers,rows):
    from build_report import table as render
    return render([tr(lang,*x) for x in headers],rows)

def picture(name,lang,z,e):
    if UNLOADED:name=name.replace('HARDWARE_COMPLETE_','HARDWARE_UNLOADED_')
    return '\n\n<!--'+name+'-->\n\n'+tr(lang,z,e)

            rows.append([node,label,f'{min(med)*factor:.2f}–{max(med)*factor:.2f}{unit}',T(f'{len(gg)}个条件的中位比值；逐条件范围见图','Condition-median ratios across '+str(len(gg))+' conditions; see figure for ranges')])
    if rows:out+=table(lang,[('节点','Node'),('观察','Observation'),('结果','Result'),('适用条件','Conditions')],rows)
    return out
