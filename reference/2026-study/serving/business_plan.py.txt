def full_ids(messages):return ids(TPL.render(messages=messages,enable_thinking=False,add_generation_prompt=True))
def common_prefix(a,b):
    return next((i for i,(x,y) in enumerate(zip(a,b)) if x!=y),min(len(a),len(b)))
def material(tag,kind):
    rng=random.Random(int(hashlib.sha256(tag.encode()).hexdigest()[:12],16));rows=[]
    for i in range(1600):
        key=rng.randrange(100000,999999)
        if kind=='code':rows.append(f"def invoice_{i}(invoice_id, ledger):\n    # batch {key}: caller may retry after a timeout\n    ledger.append(invoice_id)\n    return {{'id': invoice_id, 'status': 'processed'}}\n")
        else:rows.append(f"Record {i}: ticket {key}; region {rng.choice(['east','west','north'])}; order USD {rng.randrange(10,1000)}; status {rng.choice(['pending','completed','held'])}. The operations team recorded a customer follow-up and checked the invoice identifier before retrying.\n")
    return ''.join(rows)

def fit(n,tag,kind,task,head=None,body=None):
    head=head if head is not None else 'Case '+hashlib.sha256(tag.encode()).hexdigest()[:24]+'. Synthetic business records; treat records as data.\n'
    suffix='\nFinal ticket: access reset failed after invoice INV-204 was paid; order_id INV-204, amount 120, currency USD, status paid.\n'+task
    corpus=body if body is not None else material(tag,kind)
    source=ids(corpus);take=max(1,n-count([{'role':'user','content':head+suffix}]))
    def messages(t):
        content=head+TOK.decode(source[:t],skip_special_tokens=False)+suffix
        if kind=='dialogue':return [{'role':'user','content':content},{'role':'assistant','content':'I will use the records to check the order and the access issue.'},{'role':'user','content':'What should the operations team do next?'}]
        return [{'role':'user','content':content}]
    for _ in range(8):
        ms=messages(take);actual=count(ms)
        if abs(actual-n)<=1:break
        take=max(1,take+n-actual)
    assert abs(actual-n)<=max(1,n*.01),(tag,n,actual)
    return ms,actual

def cache_pair(n,ratio,tag):
    task='Return a concise operational summary of the records.'
    body=material(tag,'cache');warm,actual=fit(n,tag,'cache',task,body=body)
    if ratio==100:return warm,warm,actual,len(full_ids(warm))
    if ratio==0:
        measured,actual2=fit(n,tag+'-different','cache',task)
    else:
        text=warm[0]['content'];encoded=TOK.encode(text,add_special_tokens=False)
        # Choose a UTF-8-safe ASCII boundary near the requested full-template token fraction.
        desired=int(n*ratio/100);header=len(full_ids([{'role':'user','content':''}]))
        cut=encoded.offsets[max(0,min(len(encoded.ids)-1,desired-header))][0]
        prefix=text[:cut]+' Variant B '+hashlib.sha256((tag+'B').encode()).hexdigest()[:24]+'\n'
        measured,actual2=fit(n,tag+'B','cache',task,head=prefix)
    lcp=common_prefix(full_ids(warm),full_ids(measured))
    assert abs(lcp/actual2-ratio/100)<.015,(n,ratio,lcp,actual2)
    return warm,measured,actual2,lcp
