    p=ARM/'analysis/summary.json'
    if not p.exists():return ''
    d=json.loads(p.read_text());text=protocol(lang)
    text+='\n\n#### '+T('计时定义与核验','Timing definitions and validation')
    text+=table([T('指标','Metric'),T('定义与边界','Definition and scope')],[
        [T('调度器预填充耗时','Scheduler prefill duration'),T('同一调度器进程的prefill_finished_time − forward_entry_time；包括分块等待，不包括首次进入调度前的排队。','prefill_finished_time − forward_entry_time within the scheduler process; includes chunk gaps and excludes the initial admission queue.')],
        [T('预填充处理速率','Prefill processing rate'),T('同一区组Σ（输入−已缓存词元）／（最后预填充结束−最早预填充开始）。多请求重叠区间不重复累加。','Σ(input − cached tokens) / (last prefill finish − first prefill start) in the same block; overlapping request durations are not summed.')],
        [T('单请求解码速率','Per-request decode rate'),T('（实际输出−1）／（引擎总耗时−引擎首词元耗时），与原生decode_throughput交叉校验；包含引擎调度、结果传递和MTP路径。','(actual output − 1) / (engine E2E − engine TTFT), cross-checked against native decode_throughput; includes scheduling, result delivery and MTP execution.')],
        [T('并发解码窗口吞吐','Concurrent decode-window throughput'),T('Σ（实际输出−1）／（最后请求结束−最早首词元）；同一接口进程计时，并发请求的错峰预填充可能仍在此窗口内。','Σ(actual output − 1) / (last finish − first token), timed in one API process; staggered prefill admission can remain within this window.')],
        [T('MTP接受长度与接受率','MTP acceptance length and rate'),T('已安装实现的接受长度＝完成输出词元／验证次数；接受率＝正确草稿词元／提出的草稿词元。两者分母不同，接受长度不是加速倍数，也不是捕获的每一步实际输出计数。','Installed acceptance length = completed output tokens / verify count; acceptance rate = correct drafts / proposed drafts.         [T('GPU活动分解','GPU activity decomposition'),T('分别求两节点内核／拷贝区间的并集，再区分NCCL独占、其他活动独占及重叠。不相加嵌套注释，不跨节点叠加时间。','Union kernel/copy intervals separately per rank, then split NCCL-only, other-only and overlap. Nested annotations and cross-rank durations are not added.')],
    ])
    text+='\n\n'+T('调度器与接口进程的绝对时间戳转换存在偏移，因此只在同一进程的时间基准内计算区间。三个区组的中位数与最小／最大范围用于描述重复性，不据此报告尾部百分位或置信区间。发生重计算／回退的请求需单独分析，不进入当前时间拆分。',
        'Scheduler and API-process absolute timestamp conversions have different offsets; intervals are computed only within one process clock domain. Three-block medians and observed min/max ranges describe repeatability, without tail-percentile or confidence-interval claims. Retracted requests require separate analysis and are excluded from this timing decomposition.')
    if d['valid_blocks']:
        text+='\n\n#### '+T('独立速率实测','Independent rate measurements')
        text+='\n\n'+T(f"当前归档可分析{d['valid_blocks']}组、{d['valid_requests']}条正式请求；计时校验未通过{len(d['invalid'])}条。",f"The archive contains {d['valid_blocks']} analyzable blocks and {d['valid_requests']} formal requests; {len(d['invalid'])} failed timing validation.")
        text+='\n\n<!--ENGINE_INTERNAL_RATES-->\n\n'+T('引擎阶段速率｜左图为调度器预填充窗口速率；右图为单请求解码速率。阴影为区组观测范围，不是置信区间。','Engine-stage rates | Left: scheduler prefill-window rate. Right: per-request decode rate. Shading is the observed block range, not a confidence interval.')

                f"Reported cached-token fractions were {a['repeat_cached_fraction']:.0%} for immediate 8K repeats and {b['repeat_cached_fraction']:.0%} for 32K repeats.     if d['traces']:
        text+='\n\n#### '+T('GPU侧阶段与通信活动','GPU-stage and communication activity')
        text+='\n\n<!--ENGINE_GPU_TIMELINE-->\n\n'+T('两节点GPU活动分解｜各条分别归一到本节点捕获的活动跨度。剖析包含额外开销，仅用于解释路径；“其他活动”可能含未以NCCL命名的通信内核。','Per-rank GPU activity | Each bar covers its own captured GPU activity span. def figures(data):
    from plots import plt,save
    from report_visual_theme import PALETTE
    colors=PALETTE
    for lang in ['zh','en']:
        T=lambda z,e:z if lang=='zh' else e
        if data['valid_blocks']:
            fig,axes=plt.subplots(1,2,figsize=(10,4))
            for ax,stage,value,title in [(axes[0],'prefill','prefill_uncached_token_s',T('预填充：调度器窗口速率','Prefill: scheduler-window rate')),(axes[1],'decode','decode_request_median_token_s',T('解码：单请求生成速率','Decode: per-request output rate'))]:
                for i,c in enumerate([1,2,4,8]):
                    rs=sorted([g for g in data['groups'] if g['stage']==stage and g['concurrency']==c],key=lambda g:g['input'])
                    if not rs:continue
                    xs=[g['input'] for g in rs];ys=[g[value] for g in rs]
                    ax.plot(xs,ys,marker=['o','s','^','D'][i],linestyle=['-','--','-.',':'][i],color=colors[i],label=T('并发','C=')+str(c))
                    ax.fill_between(xs,[g[value+'_min'] for g in rs],[g[value+'_max'] for g in rs],color=colors[i],alpha=.12)
                ax.set_xscale('log',base=2)
                ticks=sorted({g['input'] for g in data['groups'] if g['stage']==stage})
                ax.set_xticks(ticks,[f'{n//1024}K' if n%1024==0 else f'{n:,}' for n in ticks],rotation=20)
                ax.set_ylim(bottom=0);ax.set_xlabel(T('输入词元数（1K＝1,024）','Input tokens (1K = 1,024)'));ax.set_ylabel('token/s');ax.set_title(title)
                if ax.lines:ax.legend()
            save(fig,'engine_internal_rates',lang)
        if data['traces']:
            fig,axes=plt.subplots(1,2,figsize=(10,4))
            for ax,stage in zip(axes,['prefill','decode']):
                rs=[t for t in data['traces'] if t['stage']==stage];left=[0.]*len(rs)
                for key,label,col in [('other_activity_only_ms',T('其他GPU活动','Other GPU activity'),'#004AB9'),('nccl_only_ms',T('NCCL独占','NCCL only'),'#6495ED'),('overlap_ms',T('活动重叠','Overlap'),'#A3C5FF'),('idle_gap_ms',T('未捕获活动间隙','No captured activity'),'#DBE3F0')]:
                    vals=[t[key] for t in rs];ax.barh([t['node'] for t in rs],vals,left=left,color=col,label=label);left=[a+b for a,b in zip(left,vals)]
                ax.set_title(T('预填充' if stage=='prefill' else '解码',stage.capitalize())+T('：8K／单并发',' : 8K / C=1'));ax.set_xlabel(T('GPU活动跨度（毫秒）','GPU activity span (ms)'));ax.legend(fontsize=7,loc='lower right')
            save(fig,'engine_gpu_timeline',lang)
            fig,ax=plt.subplots(figsize=(8,3.7))
            phases=['draft','verify','draft_extend']
            for i,t in enumerate([t for t in data['traces'] if t['stage']=='decode']):
                vals=[t['semantic_annotations'][k].get('mean_step_gpu_range_ms') or 0 for k in phases]
                bars=ax.bar([j+(i-.5)*.34 for j in range(3)],vals,.32,color=['#004AB9','#70A6FF'][i%2],label=t['node'])
                ax.bar_label(bars,fmt='%.2f',padding=3)
            ax.set_xticks(range(3),[T('草稿生成','Draft'),T('目标模型验证','Target verify'),T('草稿状态扩展','Draft extension')]);ax.set_ylabel(T('每步GPU关联范围（毫秒）','GPU-correlated span per step (ms)'));ax.legend();ax.set_ylim(bottom=0)
            save(fig,'engine_mtp_stages',lang)
