def results(lang):
 from build_report import table,fmt
 from length_analysis import data
 d=data();T=lambda z,e:z if lang=='zh' else e
 text='\n\n### '+T('扩展长度：覆盖、实际生成与重复性','Expanded lengths: coverage, realized output and repeatability')+'\n\n'+conclusion(lang)
 if d.get('natural_pressure'):text+='\n\n'+output_mode(lang)
 names={'calibration':('输出参数校准','Output calibration'),'input_axis':('输入轴线','Input axis'),'output_axis':('输出轴线','Output axis'),'balanced':('平衡长度','Balanced lengths'),'near_limit':('近边界','Near boundary'),'long_cache':('长前缀','Long prefixes'),'business':('业务锚点','Business anchors'),'steady_warm':('代表条件预热','Representative warmups'),'repeated_blocks':('重复区组','Repeated blocks')}
 rows=[]
 for key,(z,e) in names.items():
  rs=[r for r in d['rows'] if r['family']==key];ns=[r['actual_output'] for r in rs if r['actual_output'] is not None];rat=[r['output_attainment'] for r in rs if r['output_attainment'] is not None]
  rows.append([T(z,e),f'{sum(r["complete"] for r in rs)}/{len(rs)}',len(ns),f'{min(ns):,}–{max(ns):,}' if ns else '—',fmt(100*statistics.median(rat),1)+'%' if rat else '—',sum(r['output_attained'] for r in rs) if key!='business' else T('自然结束','Natural EOS')])
 text+=table([T('负载','Load'),T('完整／计划','Complete / planned'),T('源站计数数','Upstream counts'),T('实际输出范围','Actual output range'),T('达成率中位数','Median attainment'),T('实际长度达成数','Output length attained')],rows)
 text+='\n\n<!--LENGTH_ATTAINMENT-->\n\n'+T('输出达成｜逐档比较源站实际生成与请求上限；空白表示没有原始计数，不能按零处理。','Output attainment | Compare original upstream generation with requested caps; missing counts remain blank, not zero.')
 measured=[]
 for block in d['blocks']:
  if block['family'] not in ['input_axis','output_axis','balanced','near_limit']:continue
  rs=[r for r in d['rows'] if r['block']==block['id'] and r['output_attained'] and r['actual_input'] is not None and abs(r['input_error_fraction'])<=.01]
  if not rs:continue
  med=lambda field:statistics.median([r[field] for r in rs if r[field] is not None]) if any(r[field] is not None for r in rs) else None
  measured.append([f'{rs[0]["input_target"]//1024}K → {rs[0]["output_cap"]//1024}K',f'{len(rs)}/{block["complete"]}',fmt(med('actual_output'),0),fmt(med('ttft_s'),2),fmt(med('e2e_s'),2),fmt(1000*med('tpot_s'),1) if med('tpot_s') is not None else '—'])
 if measured:
  text+='\n\n'+T('以下仅纳入实际输入误差≤1%、实际输出达成98%–100%的完整请求。数值为中位数；单样本格就是该次观测，不输出尾分位。','The following includes only complete requests with input error ≤1% and realized output at 98–100% of cap. Values are medians; a one-sample cell is that observation, with no tail percentiles.')
  text+=table([T('输入→输出上限','Input → output cap'),T('合格／完整','Eligible / complete'),T('实际输出','Actual output'),T('首字 s','Visible TTFT s'),T('完成 s','E2E s'),T('每词元 ms（估算）','Estimated TPOT ms')],measured)
 good=[b for b in d['blocks'] if b['valid_capacity']]
 text+='\n\n'+T(f'当前 {len(good)} 个重复区组满足容量比较条件；单次边界探针不混入稳态容量。',f'{len(good)} repeat blocks currently qualify for capacity comparison; boundary probes are excluded from steady-capacity claims.')
 if good:
  text+=table([T('输入→输出／并发','Input → output / concurrency'),T('有效区组数','Eligible blocks'),T('合计吞吐 词元/s','Aggregate tokens/s'),T('区组吞吐范围','Block-rate range')],[[f'{i//1024}K → {o//1024}K / {c}',len(bs),fmt(sum(b['tokens_s']*b['elapsed_s'] for b in bs)/sum(b['elapsed_s'] for b in bs),1),fmt(min(b['tokens_s'] for b in bs),1)+'–'+fmt(max(b['tokens_s'] for b in bs),1)] for i,o,c,bs in [(i,o,c,[b for b in good if f'_i{i}_o{o}_c{c}_' in b['id']]) for i,o in [(1024,4096),(32768,1024)] for c in [1,4]] if bs])
 probes=[r for r in d['rows'] if r['family']=='long_cache' and r['cache_role']=='probe'];valid=[r for r in probes if r['cache_fraction'] is not None]
 text+='\n\n'+T(f'64K／128K前缀诊断目前有 {len(valid)}/{len(probes)} 个有效复测窗口；命中比例与目标重合分别报告。',f'The 64K/128K prefix diagnostic currently has {len(valid)}/{len(probes)} valid probe windows; observed cache fractions and intended overlap remain distinct.')
 if valid:
  text+=table([T('输入／目标重合','Input / target overlap'),T('有效／计划对','Valid / planned pairs'),T('窗口缓存比例 中位／范围','Window cache fraction median / range')],[[f'{i//1024}K / {p}%',f'{len(v)}/3',f'{statistics.median(v):.1%} / {min(v):.1%}–{max(v):.1%}'] for i,p,v in [(i,p,[r['cache_fraction'] for r in valid if r['input_target']==i and r['prefix_target']==p]) for i in [65536,131072] for p in [0,50,100]] if v])

New length findings are not extrapolated to untested concurrency or annual return.')+' [CSV](length-observations.csv) · [JSON](length_analysis.json)'
 return text
