def conclusion(lang):
 r=data();T=lambda z,e:z if lang=='zh' else e;valid=[w for w in r['windows'] if w.get('window',{}).get('valid')];n=r['supplement_completed']
 if not n:summary=T(f'完整响应 {r["complete_total"]:,} 次中，仅 {r["cache_fields"]} 次具有可用源站逐请求缓存字段，无法估计本次负载的总体请求命中率。160次专项诊断尚未形成结果；当前不确认缓存保留时长、抗逐出能力或长期稳定性。',f'Only {r["cache_fields"]} of {r["complete_total"]:,} complete responses have usable upstream per-request cache fields, so an overall request-hit fraction is not estimable. The 160-call supplement has no results yet; retention duration, eviction resistance and long-term stability remain unestablished.')
 if n:summary=T(f'缓存专项已完成 {n}/160 次调用，获得 {len(valid)} 个可归因计数窗口。结论按业务、实际空闲间隔、竞争压力与周期分别阅读；未完成或计数不满足条件的格子仍无结论。',f'The supplement has completed {n}/160 calls with {len(valid)} attributable counter windows. Interpret business, actual idle gaps, pressure and periodic reuse separately; incomplete or invalid cells remain unresolved.')
 pairs=r.get('reuse_pairs',[])
 if pairs:summary+=' '+T(f'已有 {len(pairs)} 对首次／重复观察，其中 {sum(x["reduction_s"]>0 for x in pairs)} 对重复访问首字更快；这只支持等待差异观察，尚不能确认缓存因果收益。',f'Across {len(pairs)} observed first/repeat pairs, {sum(x["reduction_s"]>0 for x in pairs)} repeats have lower TTFT. This supports a waiting-time observation, not isolated causal cache benefit.')
 bins=[b for b in r['time_bins'] if b['token_fraction'] is not None]
 if bins:
  ratio=sum(b['cached_delta'] for b in bins)/sum(b['prompt_delta'] for b in bins)
  summary+=' '+T(f'节点计数的 {len(bins)} 个有效活动窗口记录到 {ratio:.1%} 的词元加权缓存比例；这是该时段混合输入的观测值，不是逐请求命中率或长期稳定性结论。',f'Across {len(bins)} valid active node-counter windows, the token-weighted cached fraction is {ratio:.1%}. This describes the input mix in those windows, not per-request hits or long-term stability.')
 return summary
