import unittest
from gpu_extensions import conditions
from gpu_system_guard import allowance,GIB

class Protocol(unittest.TestCase):
    def test_manifest_unique_and_bounded(self):
        rows=conditions();self.assertEqual(len({c['condition_id'] for c in rows}),len(rows))
        self.assertTrue(all(c['rounds'] in [5,10] for c in rows))
        self.assertEqual({c['threads'] for c in rows if c['kind']=='scalar_fma'},{128,256,512})
        self.assertEqual({c['chains']*c['loops'] for c in rows if c['kind']=='scalar_fma'},{32768})
    def test_chase_cycle_exact_endpoints(self):
        import random
        for lines in [32,1024,32768]:
            order=list(range(lines));random.Random(20261016).shuffle(order)
            nxt={order[i]:order[(i+1)%lines] for i in range(lines)}
            x=order[0]
            for calls in range(1,9):
                for _ in range(65536):x=nxt[x]
                self.assertEqual(x,order[calls*65536%lines])
                self.assertLess(x*32,lines*32)
    def test_native_bandwidth_validation_count(self):
        for n in [1,1000,2**26,2**28]:
            workers=48*256
            expected=[max(0,(n-1-i)//workers+1) for i in range(workers)]
            self.assertEqual(sum(expected),n)
    def test_headroom_is_not_total_vram(self):
        self.assertEqual(allowance(128*GIB,16*GIB),0)
        self.assertLess(allowance(128*GIB,24*GIB),5*GIB)
    def test_shared_addresses_unique_and_bounded(self):
        for stride in [1,2,4,8,16,32]:
            slots=[(i//32)*1024+(i%32)*stride for i in range(256)]
            self.assertEqual(len(set(slots)),256);self.assertLess(max(slots),8192)
if __name__=='__main__':unittest.main()
