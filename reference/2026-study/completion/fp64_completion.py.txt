"""FP64 cuBLAS GEMM, five measured rounds and independently checked entries."""
import json
from pathlib import Path
import subprocess
import sys
import time
import torch
from gpu_system_guard import write_json

ROOT=Path(__file__).resolve().parent


def guard():
    end=time.monotonic()+600;stable=None
    while time.monotonic()<end:
        auth=json.loads((ROOT/'authorization.json').read_text())
        if time.time()>=auth['load_deadline_epoch'] or (ROOT/'STOP_GPU').exists():raise RuntimeError('hardware stop/deadline')
        p=subprocess.run(['nvidia-smi','--query-gpu=temperature.gpu','--format=csv,noheader,nounits'],capture_output=True,text=True,timeout=5)
        temp=float(p.stdout.strip()) if p.returncode==0 and p.stdout.strip() else None
        if stable is None and temp is not None and temp<85:return temp
        if temp is not None and temp<75:stable=stable or time.monotonic()
        else:stable=0
        if stable and time.monotonic()-stable>=30:return temp
        time.sleep(2)
    raise RuntimeError('thermal backoff timeout')


def run():
    torch.set_num_threads(4);torch.manual_seed(20261017);torch.cuda.set_device(0)
    rows=[]
    for n in [1024,4096,8192]:
        guard()
        a=torch.randint(-2,3,(n,n),device='cuda',dtype=torch.int8).to(torch.float64)/8
        b=torch.randint(-2,3,(n,n),device='cuda',dtype=torch.int8).to(torch.float64)/8
        c=torch.empty_like(a)
        indices=[0,n//2,n-1]
        ac=a[indices,:].cpu();bc=b[:,indices].cpu();reference=ac@bc
        for _ in range(3):torch.mm(a,b,out=c)
        torch.cuda.synchronize()
        for rep in range(1,6):
            temp=guard();start=torch.cuda.Event(enable_timing=True);stop=torch.cuda.Event(enable_timing=True)
            start.record();torch.mm(a,b,out=c);stop.record();stop.synchronize();ms=start.elapsed_time(stop)
            selected=c[indices,:][:,indices].cpu()
            err=float((selected-reference).abs().max())
            if not torch.isfinite(selected).all() or err!=0:raise RuntimeError('FP64 dyadic reference mismatch')
            row={'n':n,'precision':'FP64','round':rep,'elapsed_ms':ms,'tflops':2*n**3/(ms/1000)/1e12,
                 'validated':True,'max_abs_error':err,'checked_entries':9,'temperature_before_c':temp,
                 'backend':'torch.mm / CUDA cuBLAS; float64 inputs and output','torch_version':torch.__version__,'cuda_version':torch.version.cuda}
            rows.append(row)
            write_json(ROOT/'fp64_results.json',{'state':'running','rows':rows})
        del a,b,c;torch.cuda.empty_cache()
    write_json(ROOT/'fp64_results.json',{'state':'complete','rows':rows})


if __name__=='__main__':
    try:run()
    except BaseException as exc:
        path=ROOT/'fp64_failure.json';path.write_text(json.dumps({'state':'failed','error':type(exc).__name__+': '+str(exc)[:500]}));raise
